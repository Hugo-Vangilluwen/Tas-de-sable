\section{Cardinaux}

\begin{proposition}
Soit $G = (S, A, p)$ un graphe de tas de sable.
$$
\left| \mathcal{M}\left(G\right) \right| = \prod_{s \in \widetilde{S}} d^+(s)
$$
\end{proposition}

\begin{proof}
Par définition de la stabilité, $\forall s \in \widetilde{S},
c(s) \in [\![0;d^+(s)[\![$.
\end{proof}
