\section{Tas de sable}

Je vais maintenant présenter le modèle des tas de sable ou modèle BTW du nom de
ses trois créateurs Bak Tang Wiesenfeld en 1987. Je vais faire émerger les
différentes structures de ce modèle.

\subsection{Multi-ensembles}

\begin{definition}[Multi-ensemble]
Soit $E$ un ensemble.
Un multi-ensemble sur $E$ est une partie de $E \times \mathbb{N}$.
Intuitivement, il s'agit d'ensemble où la multiplicité de chaque élément
est spécifié.
On les notera avec des doubles accolades, par exemple $\left\{\left\{ a, a,
a, b, c, c \right\}\right\}$.

On dit que $M' \subseteq E' \times \mathbb{N}$ est un sous-multi-ensemble de
$M \subseteq E \times \mathbb{N}$ si $E' \subseteq E$ et
$\forall x \in E, m'(x) \leqslant m(x)$. On le note $M' \subseteq M$.
Le cardinal de $M$ est $|M| = \sum_{\left(x, \lambda_x\right) \in M} \lambda_x$.

On écrira pour simplifier pour $M$ un multi-ensemble sur $E$, $(F, +, f)$ un
monoïde commutatif et $\phi: E \rightarrow F$ une application :
$$
\sum_{x \in M} \phi(x) = \sum_{(x, \lambda_x) \in M} \lambda_x . \phi(x)
$$
En particulier, si $M$ est vide (i.e. $m(E) \subseteq \{0\}$) alors cette somme
vaut $f$.

Soient $M$ un multi-ensemble sur $E$, $E'$ un ensemble et
$f: E \rightarrow E'$ une application.
Pour tout $A \subseteq M$, on note le multi-ensemble image de $A$,
$\displaystyle f(A) = \left\{ (y, \lambda_y) \;|\; \lambda_y =
\sum_{(x, \lambda_x) \in A, f(x) = y} \lambda_x \right\}$.
\end{definition}

\begin{definition}[Ensemble des multi-ensembles]
On pose $\mathcal{ME}(E) = E \times \mathbb{N}$.
Il est muni de l'ordre partiel $\subseteq$.
\end{definition}

\begin{definition}[Multi-graphe]
Un multi-graphe orienté $(S, A)$ est un couple où $S$ est un ensemble fini de sommets
et $A$ une multi-partie de $S$ représentant les arêtes.
\end{definition}


\subsection{Définitions}

\begin{definition}[Graphe de tas de sable]
Un graphe des tas de sable $G$ est un triplet $(S, A, p)$ où $(S, A)$ est
un multi-graphe orienté
et $p \in S$ appelé puits qui est accessible depuis tous les sommets.
La multiplicité de $(s, u)$ indique le nombre d'arcs de $s$ à $u$.
On notera $\widetilde{S} = S \setminus \{p\}$, et pour tout $s \in S$, $V^+(s)
= \left\{ (u, \lambda) \;|\; (s, u, \lambda) \in A \right\}$ le multi-ensemble
des successeurs et $d^+(s)$ le degré sortant :
$$
d^+(s) = \left| V^+(s) \right|
$$
% $$V^+(s) = \sum_{(s, u) \in A} \pi(u)$$
\end{definition}

Dans le reste de cette partie, on fixe
$G = (S, A, p)$ un graphe de tas de sable.

\begin{definition}[Monoïde des configurations]
Une configuration sur $G$ est une fonction
$c: \widetilde{S} \rightarrow \mathbb{N}$.
Elle indique à chaque sommet le nombre de grains de sable qu'il contient.
On notera $0$ la configuration où aucun sommet ne contient de grain de sable.
Pour $s\in \widetilde{S}$, on notera $\mathbf{1}_s$ la configuration valant $1$ en $s$
et $0$ partout ailleurs.

Les configurations $\widetilde{S}^{\mathbb{N}}$  munies de la loi $+$ définie
par $(c + c')(s) = c(s) + c'(s)$ forment un monoïde abélien de neutre $0$.
Il est engendré par les configurations $\mathbf{1}_s$
pour $s \in \widetilde{S}$ :
$$
\forall c \in \widetilde{S}^{\mathbb{N}},
c = \sum_{s \in \widetilde{S}} c(s) \mathbf{1}_s
$$

Posons la configuration où tous les sommets ont leur degré sortant moins un grain :
$$
c_{max} = \sum_{s \in \widetilde{S}} \left(d^+(s) - 1\right) . \mathbf{1}_s
$$
\end{definition}

\subsection{Stabilisation}

\begin{definition}[Écroulement]
Pour une configuration $c$ et $s \in \widetilde{S}$ tels que $c(s) \geqslant
d^+(s)$, l'écroulement de $c$ en $s$ est :
$$c' = c - d^+(s) . \mathbf{1}_s + \sum_{u \in V^+(s), u \neq p} . \mathbf{1}_u$$
Un écroulement correspond à transférer un grain de sable de $s$ à ses voisins
sur chaque arc. Il ne peut être réalisé que si $c(s) \geqslant d^+(s)$.
Les grains allant dans le puits disparaissent.
\end{definition}

\begin{definition}[Stabilité]
Une configuration $c$ est dite stable si
$\forall s \in \widetilde{S}, c(s) < d^+(s)$
c'est-à-dire si aucun écoulement n'est possible.
\end{definition}

\begin{proposition}
Le résultat d'une suite d'écroulement est indépendant de l'ordre
dans lequel sont effectués les écroulements.
De plus, il existe toujours une suite
d'écroulements permettant d'obtenir une configuration stable.
Une telle configuration est unique. \\
Notons $c^\circ$ le résultat d'une telle suite appelé la stabilisation de $c$.
\end{proposition}

\begin{proof}
L'équation qui définit les écroulements montre qu'ils peuvent être réalisé dans
n'importe quel ordre.
Soit $c \in \mathbb{N}^{\widetilde{S}}$. Notons $\acute{E}(c)$ l'ensemble des
configurations obtenues à partir d'écroulements depuis $c$.
En notant $N = \sum_{s \in \widetilde{S}} c(s)$, on a $\acute{E}(c) \subseteq
[\![0;N]\!]^{\widetilde{S}}$. En particulier, $\acute{E}(c)$ est fini. \\
S'il est possible d'écrouler en un sommet $s$, après l'écroulement de n'importe
quel autre sommet, il est toujours possible d'écrouler en $s$.
Pour toutes configurations $(x, y) \in \acute{E}(c)^2$, on définissons leur
maximum $\max(x, y) \in \acute{E}(c)$ comme la configuration obtenue à partir
des écroulements de $x$ et ceux de $y$ depuis $c$ sans répéter ceux que $x$ et
$y$ ont en commun. $\acute{E}(c)$ possède une structure de semi-treillis fini.
On peut alors prendre le maximum de $\acute{E}(c)$ et poser :
$$
c^\circ = \max_{x \in \acute{E}(c)} x
$$
\end{proof}

\subsection{Structures des les tas de sable}

\begin{definition}[Monoïdes des tas de sable stables]
Notons $\circledast$ l'opération définie sur les configurations
par $c \circledast c' = (c + c')^\circ$.
Les tas de sable stables munis de la loi $\circledast$ forment un monoïde abélien de neutre $0$.

Notons $\left(\mathcal{M}(G), \circledast\right)$ ce monoïde.
\end{definition}

\begin{definition}[Groupe des tas de sable récurrents]
$(\mathcal{M}(G), \circledast)$ est un semi-groupe commutatif fini.
D'après le théorème 1, en notant $\mathcal{G}(G)$ son noyau,
$(\mathcal{G}(G), \circledast)$ est un groupe abélien fini.
Ses éléments sont appelés \textit{les tas de sable récurrents}.
\end{definition}

S'il n'y a aucune ambiguïté, on notera simplement $\mathcal{M}$ et $\mathcal{G}$.

\begin{proposition}[Neutre du groupe récurrent]
Notons $c_0$ l'élément neutre de $\mathcal{G}$.
$$
c_0 = \left( 2 c_{max} - \left(2 c_{max}\right)^\circ \right)^\circ
$$
\end{proposition}

\begin{proposition}
Soit $c \in \mathcal{M}$.
On a l'équivalence entre :
\begin{enumerate}[label=\roman*)]
    \item $c \in \mathcal{G}$
    \item $\forall x \in \mathcal{M}, \exists y \in \mathcal{G} : c = x \circledast y$
    \item $\forall x \in \mathcal{M}, \exists y \in \mathcal{M} : c = x \circledast y$
    \item $\exists y \in \widetilde{S}^{\mathbb{N}} : c = c_{max} \circledast y$
\end{enumerate}
\end{proposition}

\begin{proof}
Il s'agit d'une réécriture de la proposition 3 avec $x_0 = c_{max}$.
\end{proof}

\begin{proposition}
$$
c_0 = \left(2 c_{max} - \left(2 c_{max}\right)^\circ \right)^\circ
$$
\end{proposition}

\begin{proof}
Admis
\end{proof}

\begin{proposition}
Considérons l'expérience aléatoire où à partir de $0$, on dépose à chaque étape,
un grain de sable aléatoirement uniformément sur un sommet de $\widetilde{S}$
puis on stabilise. La probabilité que c apparaisse une infinité de fois est 1.
Ceci est la définition historique des tas de sable récurrents (d'où l'adjectif
de "récurrent").
\end{proposition}

\begin{proof}
Admis
\end{proof}
